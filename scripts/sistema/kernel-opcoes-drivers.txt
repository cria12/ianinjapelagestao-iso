# Opções do kernel para suporte "de distribuição" a hardware (lidas pelo scripts/f07-kernel.sh).
# A base é a configuração do pacote linux 6.16.1.arch1-1 do Arch Linux (fixada por commit e SHA-256
# em f07-kernel.sh): quase todos os drivers ficam como módulos (=m), carregados pelo udev.
# Aqui só se corrige o que não serve a esta ISO e se embute (=y) tudo o que o arranque live precisa,
# porque o initramfs NÃO leva módulos. Formato: OPÇÃO=y|n ('#' e linhas vazias são ignoradas).
# Justificações em docs/DECISOES.md ("Kernel: configuração base do Arch Linux e drivers em módulos").

# --- Sem informação de depuração pesada (DWARF/BTF) nem cabeçalhos embutidos
DEBUG_INFO_NONE=y
DEBUG_INFO_DWARF5=n
DEBUG_INFO_BTF=n
IKHEADERS=n
# versão do kernel exatamente 6.16.1 (os scripts e o /lib/modules usam esse nome)
LOCALVERSION_AUTO=n

# --- Módulos: sem assinatura (nem chave) e comprimidos com zstd (o kmod 34.2 do LFS tem +ZSTD)
MODULES=y
MODULE_UNLOAD=y
MODULE_SIG=n
# o LSM lockdown faz "select MODULE_SIG"; só tem efeito com Secure Boot e chaves próprias
SECURITY_LOCKDOWN_LSM=n
MODULE_COMPRESS_ZSTD=y
MODULE_COMPRESS_ALL=y
MODULE_DECOMPRESS=y

# --- Firmware: carregador com suporte a ficheiros .xz e .zst (o linux-firmware é instalado em .xz)
